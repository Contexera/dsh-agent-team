## 2. Key Technical Concepts

### 2.1 架构和权限边界

- Team是opt-in外部Cordis bundle。
- `packages/agent-team`是唯一Team authority。
- append-only operation ledger是持久事实来源。
- Session surface仅证明当前上下文，不持久维护第二套已读状态。
- 七工具：
  `team_view`、`team_inbox`、`team_attention`、`team_thread`、`team_message`、`team_dm`、`team_task`。
- Human唯一控制面Web Client，不恢复slash adapter。
- Team能力仅在`team-member` preset。
- 不修改Harness源码、agent loop或默认preset。
- typed Remote声明是输入，Typert生成物不得手改。

### 2.2 三个必须分开的状态

1. 完整事实已提供在当前上下文。
2. 显式`team_thread read`确认已读。
3. Host提交操作。

因此：
- 通知、history、message、拒绝结果可以证明完整事实。
- 它们不推进watermark、不消费marker。
- 成功回复/提交也不代替read。
- 新事实、revision、权限、Task状态每次写时仍检查。
- Main不增加Thread的`baseRevision` token。

### 2.3 已有完整Message绑定

`packages/agent-team/src/surface-visibility.ts`：

- 全输出实际文本SHA-256绑定。
- Message身份为Thread或Main Project + messageRef + bodyDigest。
- 长消息再绑定start/end/total。
- 仅当前`Session.surface.nodes`中的真实投影消息有效。
- Team notice必须有正确producer kind与`form:'notice'`。
- tool result必须沿`sourceEventSeqs`指到相同callId的`team_thread/team_message/team_task`。
- 必须以`deriveEventMessage(event)`得到的真实文本核对`textDigest`。
- 保留metadata但裁剪文本不算完整。
- 不解析正文、不做substring fallback、不相信模型复述。
- `bodyOmitted`输出不重新证明省略正文；必须依赖仍在当前surface上的原始完整输出。
- Main thread-card不证明其Thread消息或Activity。

### 2.4 长消息04已提交机制

没有新增continuation offset参数。唯一方式：
- 重复同一个`team_thread read`请求；
- 或重复同一个`team_thread message`（同address/messageRef）。

每次查当前surface最早缺失区间，返回下一段：
- 先缩小批次数量；
- 只有单条放不下才分段；
- read只有此前有效片段+本次输出覆盖全文才确认该Message；
- message始终不确认；
- 后续未读不能越过未完整Message；
- 裁剪/压缩丢片段则补缺段；
- UTF-16索引，分割时避免切开高代理项，中文/emoji真实拼接测试通过；
- 总输出32768 UTF-16 units，内部留64给最终计数/水位变化。
- 完整末段不再给“继续重复”指令。
- 容量为保守字符界限：
  `max(0, min(262144, contextWindow - 16384))`
  未知provider/model/context或NO_ADAPTER时262144。
- 不是实际token计数。
- 超容量抛明确错误，含messageRef，不确认，要求更短消息或更大上下文模型。
- history仍原有按条分页，没有总字符预算；文档明确，不宣称已实现。
- `[attachment] absolute-path`是原Host写入body的附件读取入口，不是附件正文；完整Message不等于附件已读。

export function activityText(fact: ActivityFact): string {
  if (fact.scope === 'main') {
    return `${fact.sequence} ${fact.actor} ${fact.kind} Member ${fact.memberId} · Project ${fact.projectRef} · ${fact.activityRef}`
  }
  const fields = [
    `${fact.sequence} ${fact.actor} ${fact.kind} Task ${fact.taskRef} · Thread ${fact.threadRef} · ${fact.activityRef}`,
  ]
  if (fact.claimRef !== null) fields.push(`Claim ${fact.claimRef}`)
  if (fact.claimRefs.length > 0) fields.push(`claims released ${fact.claimRefs.join(', ')}`)
  if (fact.completedClaimRefs.length > 0) fields.push(`completed claims ${fact.completedClaimRefs.join(', ')}`)
  if (fact.acceptedClaimRefs.length > 0) fields.push(`accepted claims ${fact.acceptedClaimRefs.join(', ')}`)
  if (fact.releasedClaimRefs.length > 0) fields.push(`released claims ${fact.releasedClaimRefs.join(', ')}`)
  if (fact.from !== null) fields.push(`from ${fact.from}`)
  if (fact.to !== null) fields.push(`to ${fact.to}`)
  if (fact.refollowedMembers.length > 0) {
    fields.push(`following restored for ${fact.refollowedMembers.join(', ')}`)
  }
  return fields.join(' · ')
}

export function activityFromView(

  projectRef?: string,
): ActivityFact | undefined {
  if (value.kind !== 'activity' || value.activityRef === undefined
      || value.actor === undefined || value.activity === undefined) return undefined
  if (projectRef !== undefined) {
    if (value.memberId === undefined
        || (value.activity !== 'owner-added' && value.activity !== 'owner-removed')) return undefined
    return activityFact({
      activityRef: value.activityRef, sequence: value.sequence, actor: value.actor,
      kind: value.activity, memberId: value.memberId, projectRef,
    } as AgentTeamMainActivity)
  }
  if (value.threadRef === undefined || value.taskRef === undefined) return undefined
  if (['claim','done','release'].includes(value.activity) && value.claimRef === undefined) return undefined
  if (value.activity === 'claims_released' && value.claimRefs === undefined) return undefined
  if (!['claim','done','release','claims_released','promote','accept','close','reopen','transfer'].includes(value.activity)) return undefined
  return activityFact({ ...value, kind: value.activity } as AgentTeamActivity)
}

function renderMainRejection(value: {
  projectRef?: string
  unreadCount?: number
  directCount?: number
  entries?: readonly MainEntryView[]
}) {
  const entries = value.entries ?? []
  const shown: MainEntryView[] = []
  const complete: MainEntryView[] = []
  let truncated = false
  const lines = () => {
    const canReconsider = !truncated
      && shown.length === entries.length
      && entries.length === value.unreadCount
      && entries.every(entry => entry.kind === 'message'
        ? entry.messageRef !== undefined && entry.body !== undefined
        : activityFromView(entry, value.projectRef) !== undefined)
    return [
      'Not committed — main_unread_required.',
      `${value.projectRef ?? ''} · ${value.unreadCount ?? 0} unread, ${value.directCount ?? 0} direct`,
      ...(entries.length === 0 ? [] : [
        '',
        canReconsider
          ? `${shown.length} pending Main stream entries, shown in full:`
          : `${shown.length} of ${entries.length} Main stream entries included; some content is truncated or omitted:`,
        ...shown.map(mainEntryLine),
      ]),
      recoveryFor('main_unread_required').sentence({
        projectRef: value.projectRef, completeFacts: canReconsider,
      }),
    ]
  }
  for (const entry of entries.slice(0, REJECTION_FACT_LIMIT)) {
    const cut = entry.kind === 'message'
      && (entry.body?.length ?? 0) > REJECTION_BODY_MAX_CHARACTERS
    shown.push(cut ? {
      ...entry,
      body: `${entry.body!.slice(0, REJECTION_BODY_MAX_CHARACTERS)}\n${REJECTION_BODY_TRUNCATION_MARKER}`,
    } : entry)
    truncated ||= cut
    if (lines().join('\n').length > REJECTION_MAX_CHARACTERS) {
      shown.pop()
      break
    }
    if (!cut) complete.push(entry)
  }
  return {
    text: lines().join('\n'),
    messages: complete.flatMap(entry => /* actual full messages */),
    activities: complete.flatMap(entry => {
      const fact = activityFromView(entry, value.projectRef)
      return fact === undefined ? [] : [fact]
    }),
  }
}
